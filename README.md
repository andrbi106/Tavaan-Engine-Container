## استفاده

```php
use Tavaan\Engine\Container\Container;
use Tavaan\Engine\Container\Contracts\ContainerInterface;

$app = new Container();

// ثبت سرویس‌ها
$app->singleton(LoggerInterface::class, StreamLogger::class);
$app->scoped(RequestContext::class);
$app->factory(Ulid::class, fn() => Ulid::generate());
$app->instance('config', $configRepo);
$app->alias(LoggerInterface::class, 'log');

// Resolve
$logger = $app->make(LoggerInterface::class);
$log    = $app->make('log');

// Auto-wiring (بدون explicit binding)
$obj = $app->make(MyService::class);  // وابستگی‌ها به‌صورت خودکار تزریق می‌شوند

// Override parameters
$obj = $app->make(MyService::class, ['param' => $value]);

// Contextual binding
$app->when(Mailer::class)
    ->needs('transport')
    ->give(SmtpTransport::class);

// Tagging
$app->tag([SmsChannel::class, MailChannel::class], 'notification.channel');
foreach ($app->tagged('notification.channel') as $channel) { /* ... */ }

// Method injection
$result = $app->call([$controller, 'store'], ['id' => 5]);
$result = $app->call('App\Http\Controllers\OrderController@cancel');

// Extend (Decorator)
$app->extend(CacheRepository::class, fn($repo, $app) => new LoggedCache($repo));

// Scope
$app->beginScope('request');
// ...
$app->endScope();
```
